在哪拿连接字符串：Vercel 仪表盘 → 该 Global Config → Settings / 创建 Read Token， 确认弹窗会给出完整连接字符串（形如 https://global-config.vercel.com/ecfg_xxx?token=yyy）。 令牌只显示一次，请复制保存。
⚠️ 连接字符串是只读令牌，但会出现在客户端代码里，只适合放非敏感配置（如本例的 greeting）。 

  


  

    
填入连接字符串后点上面的按钮，这里会打印获取到的全局配置。

  


  

    ③ 如何在「当前 Vue 项目」里使用（Vite 集成参考）
    
# 1) 安装官方 SDK（当前项目尚未安装）
#    npm i @vercel/global-config

# 2) 在 .env 里放连接字符串（Vite 用 VITE_ 前缀才暴露给前端）
#    VITE_GLOBAL_CONFIG="https://global-config.vercel.com/ecfg_8ocb38stza7vgsetjvuamp1kpwv6?token=xxx"

# 3) 新建 src/lib/globalConfig.js
import { createClient } from '@vercel/global-config'
const conn = import.meta.env.VITE_GLOBAL_CONFIG
const client = conn ? createClient(conn) : null
export async function getGreeting() {
  if (!client) return null
  return await client.get('greeting')   // 读单个 key
}
export async function getAllConfig() {
  if (!client) return {}
  return await client.getAll()          // 读全部
}

# 4) 在任意组件里调用
#    import { getGreeting } from '@/lib/globalConfig'
#    const greeting = await getGreeting()

# 注意：SDK 只能读，写数据要走 Vercel REST API（需 Access Token）。
