    ③ 如何在「当前 Vue 项目」里使用（Vite 集成参考）
    
# 1) 安装官方 SDK（当前项目尚未安装）
#    npm i @vercel/global-config

# 2) 在 .env 里放连接字符串（Vite 用 VITE_ 前缀才暴露给前端）
#    VITE_GLOBAL_CONFIG="https://global-config.vercel.com/ecfg_8ocb38stza7vgsetjvuamp1kpwv6?token=xxx"

# 3) 新建 src/lib/globalConfig.js
import { createClient } from '@vercel/global-config'
const conn = import.meta.env.VITE_GLOBAL_CONFIG   // Vite 不会自动暴露 process.env，必须显式传入
const client = conn ? createClient(conn) : null   // 默认 client 读 process.env.GLOBAL_CONFIG，浏览器里改用 createClient
export async function getGreeting() {
  if (!client) return null
  return await client.get('greeting')   // 读单个 key（README 主示例）
}
export async function getAllConfig() {
  if (!client) return {}
  return await client.getAll()          // 读全部；也可 getAll(['greeting','other'])
}
export async function hasGreeting() {
  if (!client) return false
  return await client.has('greeting')   // 判断 key 是否存在
}

# 4) 在任意组件里调用
#    import { getGreeting } from '@/lib/globalConfig'
#    const greeting = await getGreeting()

# 注意：
#   - SDK 只能读，写数据要走 Vercel REST API（需 Access Token）。
#   - 默认 no-store（动态获取）；静态页可用 createClient(conn,{cache:'force-cache'})（但可能读到旧值）。
#   - 返回对象不可变，改值前用 clone()。
